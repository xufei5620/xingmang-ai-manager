Write-Host "`n=== 当前 PowerShell 权限 ==="
whoami
whoami /groups | Select-String "S-1-16-"

Write-Host "`n=== 星芒进程 ==="
Get-Process -ErrorAction SilentlyContinue |
  Where-Object {
    $_.ProcessName -match "星芒|XingMang" -or
    $_.MainWindowTitle -match "星芒"
  } |
  Select-Object Id, ProcessName, Path, StartTime, MainWindowTitle |
  Format-List

Write-Host "`n=== Node.js / npm 解析结果 ==="
Get-Command node,npm -All -ErrorAction SilentlyContinue |
  Select-Object Name, CommandType, Source, Path, Version |
  Format-Table -AutoSize

Write-Host "`n=== where.exe 结果 ==="
where.exe node 2>&1
where.exe npm 2>&1

Write-Host "`n=== 实际版本 ==="
node --version 2>&1
npm --version 2>&1

Write-Host "`n=== 用户 PATH ==="
[Environment]::GetEnvironmentVariable("Path", "User")

Write-Host "`n=== 系统 PATH ==="
[Environment]::GetEnvironmentVariable("Path", "Machine")

Write-Host "`n=== 星芒记录的执行模式 ==="
$logFiles = Get-ChildItem `
  -Path $env:APPDATA,$env:LOCALAPPDATA `
  -Recurse -File -ErrorAction SilentlyContinue |
  Where-Object {
    $_.FullName -match "星芒|xingmang" -and
    $_.Extension -in ".log",".jsonl"
  } |
  Sort-Object LastWriteTime -Descending |
  Select-Object -First 20

$logFiles |
  Select-String -Pattern "cli.execution-mode|trusted-only|same-user" |
  Select-Object -Last 20 Path,LineNumber,Line |
  Format-List
